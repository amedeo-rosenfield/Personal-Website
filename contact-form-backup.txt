=== CONTACT FORM JAVASCRIPT TO SAVE ===

<script>
    // Handle visitor contact form
    const visitorForm = document.getElementById('visitorForm');
    const formSuccess = document.getElementById('formSuccess');
    const formError = document.getElementById('formError');
    
    if (visitorForm) {
        visitorForm.addEventListener('submit', function(e) {
            e.preventDefault();
            
            const formData = new FormData(visitorForm);
            const email = formData.get('email');
            const message = formData.get('message');
            
            // Show loading state
            const submitBtn = visitorForm.querySelector('.submit-btn');
            const originalText = submitBtn.innerHTML;
            submitBtn.innerHTML = '<span class="btn-text">Sending...</span>';
            submitBtn.disabled = true;
            
            // Submit to Formspree using fetch
            fetch('https://formspree.io/f/xzzvldvd', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({
                    email: email,
                    message: message
                })
            })
            .then(response => {
                if (response.ok) {
                    // Add success class to form container
                    const formContainer = visitorForm.closest('.contact-form-section');
                    formContainer.classList.add('success');
                    
                    // Change title to "Sent"
                    const title = formContainer.querySelector('h3');
                    title.textContent = 'Sent';
                    
                    // Hide form and show success message
                    visitorForm.style.display = 'none';
                    formSuccess.style.display = 'block';
                    visitorForm.reset();
                    
                    // Reset after 3 seconds
                    setTimeout(() => {
                        formContainer.classList.remove('success');
                        title.textContent = 'Drop your email';
                        visitorForm.style.display = 'block';
                        formSuccess.style.display = 'none';
                    }, 3000);
                } else {
                    // Show error message
                    formError.style.display = 'block';
                }
            })
            .catch(error => {
                // Show error message
                formError.style.display = 'block';
                console.error('Error:', error);
            })
            .finally(() => {
                // Reset button
                submitBtn.innerHTML = originalText;
                submitBtn.disabled = false;
            });
        });
    }
</script>
